下面是在 Linux（或类 Unix 系统） 中使用 C 语言编写的 select 和 poll 的完整实例代码，实现一个简单的 TCP 回显服务器（echo server），可同时处理多个客户端连接。

✅ 共同前提
编译命令（GCC）：
bash
gcc -o select_server select_server.c
gcc -o poll_server poll_server.c
测试方式：
bash
启动服务器
./select_server # 或 ./poll_server
另开终端，用 telnet 或 nc 测试
telnet 127.0.0.1 8888
或
nc 127.0.0.1 8888

一、select 示例：select_server.c

c
// select_server.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <sys/select.h>
#include <errno.h>

#define PORT 8888
#define MAX_CLIENTS 10
#define BUFFER_SIZE 1024

int main() {
int server_fd, client_fd;
struct sockaddr_in address;
int addrlen = sizeof(address);
fd_set readfds;
int max_fd = 0;

// 创建监听 socket
if ((server_fd = socket(AF_INET, SOCK_STREAM, 0)) == 0) {
perror("socket failed");
exit(EXIT_FAILURE);
}

int opt = 1;
if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt))) {
perror("setsockopt");
exit(EXIT_FAILURE);
}

address.sin_family = AF_INET;
address.sin_addr.s_addr = INADDR_ANY;
address.sin_port = htons(PORT);

if (bind(server_fd, (struct sockaddr )&address, sizeof(address)) < 0) {
perror("bind failed");
exit(EXIT_FAILURE);
}

if (listen(server_fd, 3) < 0) {
perror("listen");
exit(EXIT_FAILURE);
}

printf("Select server listening on port %d\n", PORT);

// 客户端 fd 集合
int client_fds[MAX_CLIENTS];
for (int i = 0; i < MAX_CLIENTS; i++) {
client_fds[i] = 0;
}

while (1) {
FD_ZERO(&readfds);
FD_SET(server_fd, &readfds);
max_fd = server_fd;

// 添加客户端 fd 到监控集合
for (int i = 0; i < MAX_CLIENTS; i++) {
int fd = client_fds[i];
if (fd > 0) {
FD_SET(fd, &readfds);
if (fd > max_fd) max_fd = fd;
}
}

// 阻塞等待事件
int activity = select(max_fd + 1, &readfds, NULL, NULL, NULL);
if (activity < 0 && errno != EINTR) {
perror("select error");
break;
}

// 新连接到来
if (FD_ISSET(server_fd, &readfds)) {
if ((client_fd = accept(server_fd, (struct sockaddr )&address, (socklen_t)&addrlen)) < 0) {
perror("accept");
continue;
}
printf("New connection: %d\n", client_fd);

// 加入客户端列表
for (int i = 0; i < MAX_CLIENTS; i++) {
if (client_fds[i] == 0) {
client_fds[i] = client_fd;
break;
}
}
}

// 检查客户端数据
for (int i = 0; i < MAX_CLIENTS; i++) {
int fd = client_fds[i];
if (fd > 0 && FD_ISSET(fd, &readfds)) {
char buffer[BUFFER_SIZE] = {0};
int valread = read(fd, buffer, BUFFER_SIZE);
if (valread == 0) {
// 客户端断开
printf("Client %d disconnected\n", fd);
close(fd);
client_fds[i] = 0;
} else if (valread > 0) {
// 回显
write(fd, buffer, valread);
}
}
}
}

close(server_fd);
return 0;
}

二、poll 示例：poll_server.c

c
// poll_server.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <poll.h>
#include <errno.h>

#define PORT 8888
#define MAX_CLIENTS 10
#define BUFFER_SIZE 1024

int main() {
int server_fd;
struct sockaddr_in address;
int addrlen = sizeof(address);

if ((server_fd = socket(AF_INET, SOCK_STREAM, 0)) == 0) {
perror("socket failed");
exit(EXIT_FAILURE);
}

int opt = 1;
if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt))) {
perror("setsockopt");
exit(EXIT_FAILURE);
}

address.sin_family = AF_INET;
address.sin_addr.s_addr = INADDR_ANY;
address.sin_port = htons(PORT);

if (bind(server_fd, (struct sockaddr )&address, sizeof(address)) < 0) {
perror("bind failed");
exit(EXIT_FAILURE);
}

if (listen(server_fd, 3) < 0) {
perror("listen");
exit(EXIT_FAILURE);
}

printf("Poll server listening on port %d\n", PORT);

// 使用 pollfd 数组，第一个是监听 socket
struct pollfd fds[MAX_CLIENTS + 1];
int nfds = 1;

// 初始化
fds[0].fd = server_fd;
fds[0].events = POLLIN;

for (int i = 1; i <= MAX_CLIENTS; i++) {
fds[i].fd = -1; // 无效 fd
fds[i].events = POLLIN;
}

while (1) {
int poll_count = poll(fds, nfds, -1); // 永久阻塞
if (poll_count < 0) {
if (errno == EINTR) continue;
perror("poll error");
break;
}

// 检查监听 socket
if (fds[0].revents & POLLIN) {
int client_fd = accept(server_fd, (struct sockaddr )&address, (socklen_t)&addrlen);
if (client_fd < 0) {
perror("accept");
continue;
}
printf("New connection: %d\n", client_fd);

// 找一个空位加入
int added = 0;
for (int i = 1; i <= MAX_CLIENTS; i++) {
if (fds[i].fd == -1) {
fds[i].fd = client_fd;
if (i >= nfds) nfds = i + 1;
added = 1;
break;
}
}
if (!added) {
printf("Too many clients, dropping connection\n");
close(client_fd);
}
}

// 检查客户端
for (int i = 1; i < nfds; i++) {
if (fds[i].fd == -1) continue;

if (fds[i].revents & (POLLIN POLLHUP POLLERR)) {
char buffer[BUFFER_SIZE] = {0};
int valread = read(fds[i].fd, buffer, BUFFER_SIZE);
if (valread <= 0) {
// 客户端关闭或出错
printf("Client %d disconnected\n", fds[i].fd);
close(fds[i].fd);
fds[i].fd = -1;
} else {
// 回显
write(fds[i].fd, buffer, valread);
}
}
}
}

close(server_fd);
return 0;
}

🔍 关键区别总结

特性 select poll
------ -------- -------
FD 表示 fd_set 位图 struct pollfd 数组
FD 限制 通常 ≤1024 无硬限制（受系统资源限制）
重用性 每次需重置 fd_set events 不变，只需检查 revents
错误/挂起事件 需额外处理 自动通过 POLLHUP/POLLERR 报告

⚠️ 注意事项
这些是教学级示例，未处理所有边界情况（如 partial read/write）。
实际生产中推荐使用 epoll（Linux） 或高级框架（如 libevent、libuv、asyncio）。
poll 在 macOS/BSD 上可用，但 Linux 上 epoll 性能更优。

如果你需要 epoll 版本、Python 版本的 select/poll 示例，或者想了解如何与 事件循环（如 asyncio）结合，也可以继续提问！
